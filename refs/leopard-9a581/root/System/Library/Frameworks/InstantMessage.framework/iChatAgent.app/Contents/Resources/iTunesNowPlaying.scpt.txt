tell application "iTunes"	if player state is playing then		set klass to class of current track		if klass = URL track then			set loc to address of current track		else if klass = file track then			set loc to location of current track		else			set loc to null		end if		return {name of current track, album of current track, artist of current track, finish of current track, player position, loc}	else		return null	end ifend tell
